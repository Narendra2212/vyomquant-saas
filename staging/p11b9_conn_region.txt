49: def get_redis_url(db: int = 0) -> str:
50:     """Get Redis URL for specific database.
51: 
52:     If REDIS_URL is provided, it is used as the base URL with the DB number appended.
53:     Otherwise, constructs URL from REDIS_HOST, REDIS_PORT, REDIS_PASSWORD.
54: 
55:     Supports both redis:// and rediss:// schemes for TLS.
56:     """
57:     if REDIS_URL:
58:         # Parse existing REDIS_URL and append DB number
59:         # Handle URLs with existing path or query parameters
60:         from urllib.parse import urlparse, urlunparse, parse_qs, urlencode
61: 
62:         parsed = urlparse(REDIS_URL)
63: 
64:         # Extract existing query parameters
65:         query_params = parse_qs(parsed.query)
66: 
67:         # Set DB via query parameter or path
68:         # Redis-py supports both ?db=0 and /0 syntax
69:         query_params['db'] = str(db)
70: 
71:         # Reconstruct URL with new DB parameter
72:         new_query = urlencode(query_params, doseq=True)
73:         new_url = urlunparse((
74:             parsed.scheme,
75:             parsed.netloc,
76:             parsed.path,
77:             parsed.params,
78:             new_query,
79:             parsed.fragment
80:         ))
81: 
82:         return new_url
83:     else:
84:         # Fallback to host/port/password construction
85:         auth = f":{REDIS_PASSWORD}@" if REDIS_PASSWORD else ""
86:         return f"redis://{auth}{REDIS_HOST}:{REDIS_PORT}/{db}"
